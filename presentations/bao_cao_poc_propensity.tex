\begin{frame}{Mục tiêu và phạm vi bài toán}
\begin{itemize}
  \item Ước lượng xác suất household phát sinh ít nhất một basket được ghi nhận trong 7 ngày sau cutoff.
  \item Tình huống giả định: retailer cần thông tin hỗ trợ dự báo hoặc sắp xếp ưu tiên household.
  \item Đơn vị phân tích: một household tại một cutoff; bao gồm fuel, miscellaneous và basket zero-sales.
  \item Kết quả dự báo hành vi được ghi nhận, chưa đo tác động của coupon hoặc liên hệ.
\end{itemize}
\end{frame}

\begin{frame}{Tổng quan dữ liệu Dunnhumby}
\begin{itemize}
  \item 2.595.732 dòng transaction; 276.484 basket; 2.500 household; chỉ số ngày 1--711.
  \item Một dòng transaction là một sản phẩm trong basket; không đồng nhất số dòng với số chuyến mua.
  \item Bảng bổ trợ: product, demographic, campaign, coupon và product--store--week.
  \item Có demographic cho 801 household; thời điểm thông tin sẵn có trong lịch sử chưa được xác minh.
\end{itemize}
\end{frame}

\begin{frame}{Chất lượng dữ liệu và giới hạn}
\begin{itemize}
  \item Hai ngày 278 và 643 có một basket mỗi ngày; nguyên nhân chưa được xác nhận.
  \item Không có ánh xạ đáng tin cậy từ \texttt{DAY} sang ngày lịch thực.
  \item Giữ \texttt{QUANTITY} trong nguồn nhưng không tổng hợp xuyên department do chưa rõ đơn vị.
  \item Demographic và marketing chưa dùng trong pipeline POC vì chưa rõ thời điểm sẵn có.
  \item Một household có thể xuất hiện tại nhiều cutoff; các quan sát không độc lập.
\end{itemize}
\end{frame}

\begin{frame}{Thiết kế household--snapshot}
\begin{itemize}
  \item Tại cutoff $s$, dùng lịch sử đến hết $s$; target xét basket trong $(s,s+7]$.
  \item Chỉ xét household đã được ghi nhận đến cutoff; tạo snapshot theo tuần.
  \item Feature cơ bản: recency, thời gian quan sát và lịch sử basket, sales, số dòng, ngày hoạt động.
  \item Cửa sổ lịch sử 7, 30, 90 và 180 ngày; không dùng thông tin sau cutoff.
\end{itemize}
\end{frame}

\begin{frame}{Baseline và mô hình}
\begin{itemize}
  \item Baseline: xác suất mua theo sáu nhóm recency, ước lượng trên train.
  \item Mô hình tham chiếu: Logistic Regression; ứng viên phi tuyến: LightGBM.
  \item Thử riêng chu kỳ mua, thay đổi hoạt động, thành phần basket, discount và kích thước chuyến mua.
  \item Chu kỳ mua cải thiện Brier và log loss trong bốn fold backtest, nhưng lift chưa tăng đều.
\end{itemize}
\end{frame}

\begin{frame}{Calibration và quy trình đánh giá POC}
\begin{itemize}
  \item LightGBM fit trên cutoff 180--439; sigmoid fit trên 453--481; chọn calibration trên 495--537.
  \item Quy tắc định trước: giữ sigmoid nếu log loss giảm ít nhất 0,001 và Brier không tăng.
  \item Cả hai phương án giữ xác suất thô; không refit model sau bước chọn calibration.
  \item Ứng viên chính: 18 feature + chu kỳ; 18 feature là phương án đối chứng.
\end{itemize}
\end{frame}

  \item Lift tính top 10\% từng cutoff, đồng điểm được xử lý bằng kỳ vọng chọn ngẫu nhiên.
\end{itemize}
\end{frame}

\begin{frame}{Kịch bản xếp hạng trên validation}
\begin{itemize}
  \item Ở năng lực 10\%, có 2.750 lượt household--cutoff qua 11 cutoff.
  \item Tỷ lệ mua ghi nhận: 96,58\% theo LightGBM + chu kỳ, so với 70,82\% kỳ vọng theo recency.
  \item Hai phương pháp dùng cùng giai đoạn fit đến cutoff 439; baseline recency có nhiều đồng điểm.
  \item Chênh lệch phản ánh xếp hạng, không phải giao dịch tăng thêm do can thiệp.
\end{itemize}
\end{frame}

\begin{frame}{Deployment giả lập và kết luận}
\begin{itemize}
  \item Quy trình: dữ liệu đến cutoff $\rightarrow$ tính feature $\rightarrow$ pipeline đã lưu $\rightarrow$ xác suất và thứ hạng.
  \item Cutoff 711 tạo 2.500 dự báo cho ngày 712--718; nguồn không có nhãn để kiểm chứng.
  \item Cần theo dõi chất lượng nguồn, phân bố feature, calibration và hiệu quả theo thời gian khi có nhãn.
  \item POC cho thấy tín hiệu dự báo; hiệu quả kinh doanh và khả năng tổng quát hóa cần xác minh thêm.
\end{itemize}
\end{frame}
